// Tu URL de Render bien configurada const URL_MI_API = "https://gnartejapi.onrender.com"; const botonEnviar = document.getElementById('boton-enviar'); // Cambia el ID si el tuyo se llama diferente const inputMensaje = document.getElementById('input-mensaje'); const contenedorChat = document.getElementById('contenedor-chat'); // Función para enviar el mensaje al servidor de Render async function enviarMensajeAlChat() { const texto = inputMensaje.value.trim(); if (!texto) return; // 1. Pintar tu mensaje en la pantalla agregarMensajeAlDiseño(texto, 'usuario'); inputMensaje.value = ''; try { // 2. Llamada mágica a TU servidor de Render const respuesta = await fetch(URL_MI_API + "/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ mensaje: texto }) }); if (!respuesta.ok) { throw new Error("Error en el servidor"); } const data = await respuesta.json(); // 3. Pintar la respuesta de tu IA en la pantalla agregarMensajeAlDiseño(data.respuesta, 'bot'); } catch (error) { console.error("Hubo un corte de cables:", error); agregarMensajeAlDiseño("Lo siento, no he podido conectar con el servidor.", 'error'); } } // Función auxiliar para ir pintando los bocadillos del chat function agregarMensajeAlDiseño(texto, emisor) { const div = document.createElement('div'); div.classList.add('mensaje', emisor); div.innerText = texto; contenedorChat.appendChild(div); contenedorChat.scrollTop = contenedorChat.scrollHeight; // Auto-scroll hacia abajo } // Escuchar el clic del botón y la tecla Enter botonEnviar.addEventListener('click', enviarMensajeAlChat); inputMensaje.addEventListener('keypress', function(e) { if (e.key === 'Enter') enviarMensajeAlChat(); });
